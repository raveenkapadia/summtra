What are Parans?
Unlike vertical AC/DC/MC/IC lines that only pass through specific longitudes, PARAN lines are latitude-based and exist at EVERY latitude. They show where two planets share angular relationships.
Why Parans Matter:
Even if no direct lines pass through India, every Indian city HAS paran influences at its latitude!

javascript// For each city's latitude, find planets with aligned angles
function calculateParans(birthData, cityLatitude) {
  const parans = [];
  const planets = ['Sun', 'Moon', 'Mercury', 'Venus', 'Mars', 'Jupiter', 'Saturn'];
  
  for (let i = 0; i < planets.length; i++) {
    for (let j = i + 1; j < planets.length; j++) {
      // Check if both planets are near horizon or meridian at this latitude
      const alignment = checkAngularAlignment(birthData, planets[i], planets[j], cityLatitude);
      if (alignment.isAligned) {
        parans.push({
          planet1: planets[i],
          planet2: planets[j],
          type: alignment.type, // 'rising-rising', 'rising-culminating', etc.
          strength: alignment.strength
        });
      }
    }
  }
  return parans;
}

javascriptfunction calculateCityScore(cityData, birthData, goal) {
  
  // WESTERN ASTROCARTOGRAPHY (50 points)
  
  // Line Proximity - 25 points max
  const lineDistance = calculateDistanceToNearestBeneficialLine(cityData, birthData, goal);
  let lineScore;
  if (lineDistance < 100) lineScore = 25;
  else if (lineDistance < 200) lineScore = 22;
  else if (lineDistance < 350) lineScore = 18;
  else if (lineDistance < 500) lineScore = 14;
  else if (lineDistance < 700) lineScore = 10;
  else lineScore = 5;
  
  // Paran Lines - 25 points max
  const parans = calculateParans(birthData, cityData.latitude);
  const beneficialParans = filterBeneficialParansForGoal(parans, goal);
  let paranScore;
  if (beneficialParans.length >= 3) paranScore = 25;
  else if (beneficialParans.length === 2) paranScore = 20;
  else if (beneficialParans.length === 1) paranScore = 15;
  else paranScore = 8;
  
  // VEDIC ASTROLOGY (50 points)
  
  // Nakshatra + Rashi - 20 points max
  const directionFromBirth = calculateDirection(birthData.birthPlace, cityData);
  const nakshatraMatch = isDirectionFavorableForNakshatra(birthData.nakshatra, directionFromBirth);
  const rashiMatch = isRashiCompatibleWithCity(birthData.rashi, cityData);
  const nakshatraRashiScore = (nakshatraMatch ? 12 : 4) + (rashiMatch ? 8 : 3);
  
  // Lagna-Vastu - 15 points max
  const lagnaDirection = getLagnaFavorableDirection(birthData.lagna);
  const vastuMatch = isVastuAligned(lagnaDirection, directionFromBirth);
  const lagnaVastuScore = vastuMatch ? 15 : 7;
  
  // Dasha Timing - 15 points max
  const currentDasha = getCurrentDasha(birthData);
  const dashaLineMatch = doesDashaPlanetHaveLineOrParan(currentDasha, cityData, parans);
  const dashaScore = dashaLineMatch ? 15 : 8;
  
  // TOTAL
  const totalScore = lineScore + paranScore + nakshatraRashiScore + lagnaVastuScore + dashaScore;
  
  return {
    total: totalScore,
    breakdown: {
      western: {
        lineProximity: { score: lineScore, max: 25, distance: lineDistance },
        parans: { score: paranScore, max: 25, details: beneficialParans }
      },
      vedic: {
        nakshatraRashi: { score: nakshatraRashiScore, max: 20, direction: directionFromBirth },
        lagnaVastu: { score: lagnaVastuScore, max: 15, aligned: vastuMatch },
        dashaTiming: { score: dashaScore, max: 15, planet: currentDasha.planet }
      }
    }
  };
}

#### 2. Scoring Logic (confirm this matches PDF display)
```javascript
function calculateCityScore(city, birthData, goal) {
  
  // === WESTERN ASTROCARTOGRAPHY (50 points max) ===
  
  // Line Proximity (25 points max)
  // Based on distance to nearest relevant planetary line
  let lineProximityScore = 0;
  const nearestLineDistance = getNearestLineDistance(city, birthData);
  
  if (nearestLineDistance <= 100) lineProximityScore = 25;      // Direct hit
  else if (nearestLineDistance <= 350) lineProximityScore = 22; // Strong orb
  else if (nearestLineDistance <= 700) lineProximityScore = 18; // Moderate orb
  else if (nearestLineDistance <= 1000) lineProximityScore = 15; // Weak orb
  else lineProximityScore = 10; // Minimal influence (baseline)
  
  // Paran Lines (25 points max)
  // Based on latitude-based planetary alignments
  const activeParans = getParanLinesAtLatitude(city.lat, birthData);
  const relevantParans = filterParansForGoal(activeParans, goal);
  let paranScore = Math.min(25, 10 + (relevantParans.length * 5)); // 10 base + 5 per paran
  
  const westernTotal = lineProximityScore + paranScore;
  
  // === VEDIC ASTROLOGY (50 points max) ===
  
  // Nakshatra + Rashi Compatibility (20 points max)
  const nakshatraDirection = getFavorableDirection(birthData.nakshatra);
  const cityDirection = getDirection(birthData.birthPlace, city);
  const directionMatch = isDirectionCompatible(nakshatraDirection, cityDirection);
  let nakshatraScore = directionMatch ? 20 : 10;
  
  // Lagna-Vastu Alignment (15 points max)
  const lagnaFavorableDirections = getVastuDirections(birthData.lagna);
  const vastuMatch = lagnaFavorableDirections.includes(cityDirection);
  let lagnaVastuScore = vastuMatch ? 15 : 8;
  
  // Dasha Timing (15 points max)
  const currentDasha = birthData.dasha.mahadasha;
  const dashaRelevantLines = getDashaLines(currentDasha); // e.g., Mercury dasha → Mercury lines
  const hasDashaLine = cityHasLineInfluence(city, dashaRelevantLines);
  let dashaScore = hasDashaLine ? 15 : 10;
  
  const vedicTotal = nakshatraScore + lagnaVastuScore + dashaScore;
  
  // === FINAL SCORE ===
  const totalScore = westernTotal + vedicTotal;
  const percentage = totalScore; // Already out of 100
  
  return {
    western: {
      lineProximity: { score: lineProximityScore, max: 25 },
      paranLines: { score: paranScore, max: 25 },
      total: westernTotal
    },
    vedic: {
      nakshatraRashi: { score: nakshatraScore, max: 20 },
      lagnaVastu: { score: lagnaVastuScore, max: 15 },
      dashaTiming: { score: dashaScore, max: 15 },
      total: vedicTotal
    },
    finalPercentage: percentage // THIS is what we display
  };
}
```

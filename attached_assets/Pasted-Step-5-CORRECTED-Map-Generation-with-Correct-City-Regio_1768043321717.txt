City Selection Logic for Reports
javascriptconst ALL_GOALS = ['Career', 'Wealth', 'Love', 'Education', 'Settlement'];

// City counts per report type
const CITY_COUNTS = {
  single: {
    india: { best: 9, avoid: 5 },        // From 31 available
    international: { best: 9, avoid: 5 }, // From 57 available
    both: { best: 18, avoid: 10 }         // 9+9 best, 5+5 avoid
  },
  complete: {
    // Per goal counts (same city may appear in multiple goals)
    india: { bestPerGoal: 9, avoidPerGoal: 5 },
    international: { bestPerGoal: 9, avoidPerGoal: 5 },
    both: { bestPerGoal: 18, avoidPerGoal: 10 }
  }
};

class ReportGenerator {
  constructor() {
    this.mapRenderer = new AstroMapRenderer();
    this.legendRenderer = new LegendRenderer();
    this.tempDir = path.join(__dirname, '../temp/maps');
    
    if (!fs.existsSync(this.tempDir)) {
      fs.mkdirSync(this.tempDir, { recursive: true });
    }
  }
  
  // Get regions based on scope
  getRegionsForScope(scope) {
    if (scope === 'india') return INDIA_REGIONS;
    if (scope === 'international') return INTERNATIONAL_REGIONS;
    return [...INDIA_REGIONS, ...INTERNATIONAL_REGIONS]; // both
  }
  
  // Get city counts based on report type and scope
  getCityCounts(reportType, scope) {
    return CITY_COUNTS[reportType][scope];
  }
  
  // Filter cities by scope
  filterCitiesByScope(cities, scope) {
    if (scope === 'india') {
      return cities.filter(c => c.country === 'India');
    } else if (scope === 'international') {
      return cities.filter(c => c.country !== 'India');
    }
    return cities; // both
  }
  
  // Get top cities for a goal, sorted by score
  getTopCitiesForGoal(allCities, goal, scope, limit) {
    const filtered = this.filterCitiesByScope(allCities, scope);
    
    return filtered
      .sort((a, b) => {
        const scoreA = a.goalScores?.[goal] || a.score || 0;
        const scoreB = b.goalScores?.[goal] || b.score || 0;
        return scoreB - scoreA;
      })
      .slice(0, limit);
  }
  
  // Get cities to avoid for a goal (lowest scores)
  getAvoidCitiesForGoal(allCities, goal, scope, limit) {
    const filtered = this.filterCitiesByScope(allCities, scope);
    
    return filtered
      .sort((a, b) => {
        const scoreA = a.goalScores?.[goal] || a.score || 0;
        const scoreB = b.goalScores?.[goal] || b.score || 0;
        return scoreA - scoreB;
      })
      .slice(0, limit);
  }
  
  // Get cities for a specific region
  getCitiesForRegion(allCities, regionKey) {
    const regionMapping = {
      'india_north': 'North India',
      'india_south': 'South India',
      'india_west': 'West India',
      'india_east': 'East India',
      'india_central': 'Central India',
      'europe': 'Europe',
      'middle_east': 'Middle East',
      'southeast_asia': 'Southeast Asia',
      'east_asia': 'East Asia',
      'north_america': 'North America',
      'south_america': 'South America',
      'australia': 'Australia & Oceania',
      'africa': 'Africa'
    };
    
    const regionName = regionMapping[regionKey];
    return allCities.filter(c => c.region === regionName);
  }
  
  // Main function: Generate all maps for report
  async generateReportMaps(userData, astroData, options) {
    const { goal, scope, reportType, birthLocation } = options;
    const reportId = `report_${Date.now()}`;
    
    const maps = {
      overview: {},
      regional: {},
      cities: {},
      avoidCities: {},
      powerZones: {},
      legend: null
    };
    
    const goals = reportType === 'complete' ? ALL_GOALS : [goal];
    const regions = this.getRegionsForScope(scope);
    const cityCounts = this.getCityCounts(reportType, scope);
    
    console.log(`Generating maps for ${reportType} report, ${scope} scope, goals: ${goals.join(', ')}`);
    
    // 1. OVERVIEW MAPS (per goal)
    for (const g of goals) {
      // World with all lines (reference)
      maps.overview[`${g}_all_lines`] = await this.generateMap(reportId, `${g}_all_lines`, {
        viewType: 'world',
        goal: 'Complete',
        lines: astroData.lines,
        cities: [],
        powerZones: [],
        birthLocation
      });
      
      // World filtered for this goal
      maps.overview[`${g}_filtered`] = await this.generateMap(reportId, `${g}_filtered`, {
        viewType: 'world',
        goal: g,
        lines: astroData.lines,
        cities: this.getTopCitiesForGoal(astroData.cities, g, scope, cityCounts.best || cityCounts.bestPerGoal),
        powerZones: astroData.powerZones?.filter(z => z.category?.toLowerCase() === g.toLowerCase()) || [],
        birthLocation
      });
      
      // Power zones for this goal
      maps.powerZones[g] = await this.generateMap(reportId, `${g}_power`, {
        viewType: 'world',
        goal: g,
        lines: astroData.lines,
        cities: [],
        powerZones: astroData.powerZones || [],
        birthLocation,
        highlightPowerZones: true
      });
    }
    
    // 2. REGIONAL MAPS (per goal, per region)
    for (const g of goals) {
      maps.regional[g] = {};
      
      for (const region of regions) {
        const regionCities = this.getCitiesForRegion(astroData.cities, region.key);
        
        if (regionCities.length > 0 || region.key === 'world' || region.key === 'india') {
          maps.regional[g][region.key] = await this.generateMap(reportId, `${g}_${region.key}`, {
            viewType: region.viewType,
            goal: g,
            lines: astroData.lines,
            cities: regionCities.map(c => ({ ...c, showLabel: true })),
            powerZones: astroData.powerZones || [],
            birthLocation
          });
        }
      }
    }
    
    // 3. CITY CLOSE-UP MAPS (Best Cities per goal)
    for (const g of goals) {
      maps.cities[g] = {};
      
      const topCities = this.getTopCitiesForGoal(
        astroData.cities,
        g,
        scope,
        cityCounts.best || cityCounts.bestPerGoal
      );
      
      for (const city of topCities) {
        const cityKey = city.name.toLowerCase().replace(/\s+/g, '_');
        
        maps.cities[g][cityKey] = await this.generateCityMap(reportId, city, {
          goal: g,
          lines: astroData.lines,
          powerZones: astroData.powerZones || [],
          birthLocation,
          isAvoidCity: false
        });
      }
    }
    
    // 4. AVOID CITY MAPS (per goal)
    for (const g of goals) {
      maps.avoidCities[g] = {};
      
      const avoidCities = this.getAvoidCitiesForGoal(
        astroData.cities,
        g,
        scope,
        cityCounts.avoid || cityCounts.avoidPerGoal
      );
      
      for (const city of avoidCities) {
        const cityKey = city.name.toLowerCase().replace(/\s+/g, '_');
        
        maps.avoidCities[g][cityKey] = await this.generateCityMap(reportId, city, {
          goal: g,
          lines: astroData.lines,
          powerZones: astroData.powerZones || [],
          birthLocation,
          isAvoidCity: true
        });
      }
    }
    
    // 5. LEGEND PAGE
    maps.legend = await this.generateLegend(reportId);
    
    // Log summary
    const mapCount = this.countMaps(maps);
    console.log(`Generated ${mapCount} maps for report ${reportId}`);
    
    return { reportId, maps, mapCount };
  }
  
  // Count total maps generated
  countMaps(maps) {
    let count = 0;
    count += Object.keys(maps.overview).length;
    count += Object.values(maps.regional).reduce((sum, goalMaps) => sum + Object.keys(goalMaps).length, 0);
    count += Object.values(maps.cities).reduce((sum, goalMaps) => sum + Object.keys(goalMaps).length, 0);
    count += Object.values(maps.avoidCities).reduce((sum, goalMaps) => sum + Object.keys(goalMaps).length, 0);
    count += Object.keys(maps.powerZones).length;
    count += maps.legend ? 1 : 0;
    return count;
  }
  
  // Generate single map
  async generateMap(reportId, mapName, options) {
    const filename = `${reportId}_${mapName}.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = await this.mapRenderer.renderGoalFilteredMap(options);
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`
    };
  }
  
  // Generate city close-up map
  async generateCityMap(reportId, city, options) {
    const { getCityView } = require('./map-renderer');
    const viewConfig = getCityView(city);
    
    const filename = `${reportId}_city_${city.name.toLowerCase().replace(/\s+/g, '_')}.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = await this.mapRenderer.renderGoalFilteredMap({
      ...options,
      viewConfig: viewConfig,
      cities: [{
        ...city,
        type: options.isAvoidCity ? 'avoid' : 'recommended',
        showLabel: true
      }]
    });
    
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`,
      cityName: city.name,
      isAvoidCity: options.isAvoidCity
    };
  }
  
  // Generate legend page
  async generateLegend(reportId) {
    const filename = `${reportId}_legend.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = this.legendRenderer.render();
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`
    };
  }
  
  // Cleanup temp files
  cleanupTempFiles(reportId) {
    const files = fs.readdirSync(this.tempDir);
    files.forEach(file => {
      if (file.startsWith(reportId)) {
        fs.unlinkSync(path.join(this.tempDir, file));
      }
    });
  }
}

module.exports = ReportGenerator;

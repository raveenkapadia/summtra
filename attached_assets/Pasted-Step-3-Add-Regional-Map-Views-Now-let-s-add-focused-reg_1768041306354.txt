Add City Close-up View Generator
javascript// Generate city-focused view configuration
function getCityView(city) {
  return {
    projection: 'geoMercator',
    scale: 4000,  // Very zoomed in
    center: [city.longitude, city.latitude],
    bounds: [
      [city.longitude - 5, city.latitude - 3],
      [city.longitude + 5, city.latitude + 3]
    ],
    label: city.name
  };
}

// Generate view for multiple cities (cluster view)
function getClusterView(cities) {
  const lngs = cities.map(c => c.longitude);
  const lats = cities.map(c => c.latitude);
  
  const centerLng = (Math.min(...lngs) + Math.max(...lngs)) / 2;
  const centerLat = (Math.min(...lats) + Math.max(...lats)) / 2;
  
  const lngSpread = Math.max(...lngs) - Math.min(...lngs);
  const latSpread = Math.max(...lats) - Math.min(...lats);
  const maxSpread = Math.max(lngSpread, latSpread);
  
  // Calculate scale based on spread
  const scale = Math.min(3000, Math.max(800, 15000 / (maxSpread + 5)));
  
  return {
    projection: 'geoMercator',
    scale: scale,
    center: [centerLng, centerLat],
    bounds: [
      [Math.min(...lngs) - 2, Math.min(...lats) - 2],
      [Math.max(...lngs) + 2, Math.max(...lats) + 2]
    ],
    label: 'Regional View'
  };
}

module.exports = { REGIONAL_VIEWS, getCityView, getClusterView };
Update getProjection Method
javascriptgetProjection(viewConfig) {
  const { projection, scale, center } = viewConfig;
  
  let proj;
  switch(projection) {
    case 'geoMercator':
      proj = d3.geoMercator();
      break;
    case 'geoNaturalEarth1':
      proj = d3.geoNaturalEarth1();
      break;
    case 'geoEquirectangular':
      proj = d3.geoEquirectangular();
      break;
    default:
      proj = d3.geoNaturalEarth1();
  }
  
  return proj
    .scale(scale)
    .center(center)
    .translate([MAP_WIDTH / 2, MAP_HEIGHT / 2]);
}
Add Region Label to Map
javascriptdrawRegionLabel(label) {
  if (!label) return;
  
  // Semi-transparent background
  this.ctx.fillStyle = 'rgba(45, 27, 78, 0.8)';
  this.ctx.fillRect(20, 20, label.length * 12 + 30, 35);
  
  // Border
  this.ctx.strokeStyle = '#D4AF37';
  this.ctx.lineWidth = 1;
  this.ctx.strokeRect(20, 20, label.length * 12 + 30, 35);
  
  // Text
  this.ctx.fillStyle = '#D4AF37';
  this.ctx.font = 'bold 16px Raleway, sans-serif';
  this.ctx.fillText(label, 35, 43);
}

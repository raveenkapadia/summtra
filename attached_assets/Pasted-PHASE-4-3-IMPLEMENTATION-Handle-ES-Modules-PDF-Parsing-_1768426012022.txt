```javascript
// test/templateDataFlow.test.js

describe('Template Data Flow', () => {
  let prepareReportData;
  
  beforeAll(async () => {
    // Dynamic import for ES modules
    const module = await import('../src/templateProcessor.js');
    prepareReportData = module.prepareReportData;

  test('LAGNA_ARTICLE exists for Aquarius', async () => {
    const data = await prepareReportData({ lagna: 'Aquarius', ... });
```

CHALLENGE 2: PDF Text Extraction
═══════════════════════════════════════════════════════════════

Step 1: Install pdf-parse
```bash
npm install pdf-parse --save-dev
```

Step 2: Create helper utility
```javascript
// test/helpers/pdfHelper.js

const fs = require('fs');
const pdfParse = require('pdf-parse');

async function extractTextFromPDF(pdfPath) {
  const dataBuffer = fs.readFileSync(pdfPath);
  const data = await pdfParse(dataBuffer);
  return data.text;
}

module.exports = { extractTextFromPDF };
```

Step 3: Use in tests
```javascript
// test/pdfContentVerification.test.js

const { extractTextFromPDF } = require('./helpers/pdfHelper');

describe('PDF Content', () => {
  test('Aquarius shows "as an Aquarius rising"', async () => {
    // Generate or use existing test PDF
    const pdfPath = './test-pdfs/aquarius-test.pdf';
    const text = await extractTextFromPDF(pdfPath);
    
    expect(text).toContain('as an Aquarius rising');
  });
});
```

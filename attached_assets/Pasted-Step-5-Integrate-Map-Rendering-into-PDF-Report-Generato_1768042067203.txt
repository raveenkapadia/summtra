Step 5: Integrate Map Rendering into PDF Report Generator
Now let's connect our map renderer to the PDF report generation system. We need to generate map images and embed them in the HTML templates that Puppeteer converts to PDF.
Update Report Generator
Update server/reportGenerator.js to include map generation:
javascriptconst AstroMapRenderer = require('./map-renderer');
const LegendRenderer = require('./legend-renderer');
const fs = require('fs');
const path = require('path');

class ReportGenerator {
  constructor() {
    this.mapRenderer = new AstroMapRenderer();
    this.legendRenderer = new LegendRenderer();
    this.tempDir = path.join(__dirname, '../temp/maps');
    
    // Ensure temp directory exists
    if (!fs.existsSync(this.tempDir)) {
      fs.mkdirSync(this.tempDir, { recursive: true });
    }
  }
  
  // Generate all maps needed for report
  async generateReportMaps(userData, astroData, options) {
    const { goal, scope, birthLocation } = options;
    const reportId = `report_${Date.now()}`;
    const maps = {};
    
    // 1. World Overview Map (All Lines)
    maps.worldOverview = await this.generateMap(reportId, 'world_overview', {
      viewType: 'world',
      goal: 'Complete',  // Show all lines
      lines: astroData.lines,
      cities: astroData.topCities,
      powerZones: astroData.powerZones,
      birthLocation: birthLocation
    });
    
    // 2. World Map - Goal Filtered
    maps.worldGoalFiltered = await this.generateMap(reportId, 'world_goal', {
      viewType: 'world',
      goal: goal,
      lines: astroData.lines,
      cities: astroData.topCities.filter(c => c.goalMatch === goal),
      powerZones: astroData.powerZones.filter(z => z.category === goal.toLowerCase()),
      birthLocation: birthLocation
    });
    
    // 3. Power Zones Map
    maps.powerZones = await this.generateMap(reportId, 'power_zones', {
      viewType: 'world',
      goal: goal,
      lines: astroData.lines,
      cities: [],
      powerZones: astroData.powerZones,
      birthLocation: birthLocation,
      highlightPowerZones: true
    });
    
    // 4. Regional Maps based on scope
    if (scope === 'india' || scope === 'both') {
      maps.india = await this.generateMap(reportId, 'india', {
        viewType: 'india',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.country === 'India'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      // Sub-regions
      maps.indiaNorth = await this.generateMap(reportId, 'india_north', {
        viewType: 'india_north',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'North India'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      maps.indiaSouth = await this.generateMap(reportId, 'india_south', {
        viewType: 'india_south',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'South India'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      maps.indiaWest = await this.generateMap(reportId, 'india_west', {
        viewType: 'india_west',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'West India'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      maps.indiaEast = await this.generateMap(reportId, 'india_east', {
        viewType: 'india_east',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'East India'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
    }
    
    if (scope === 'international' || scope === 'both') {
      maps.europe = await this.generateMap(reportId, 'europe', {
        viewType: 'europe',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'Europe'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      maps.middleEast = await this.generateMap(reportId, 'middle_east', {
        viewType: 'middle_east',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'Middle East'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
      
      maps.southeastAsia = await this.generateMap(reportId, 'southeast_asia', {
        viewType: 'southeast_asia',
        goal: goal,
        lines: astroData.lines,
        cities: astroData.topCities.filter(c => c.region === 'Southeast Asia'),
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
    }
    
    // 5. City Close-up Maps (for top recommended cities)
    maps.cityMaps = {};
    const topCities = astroData.topCities.slice(0, 9); // Top 6-9 cities
    
    for (const city of topCities) {
      const cityKey = city.name.toLowerCase().replace(/\s+/g, '_');
      maps.cityMaps[cityKey] = await this.generateCityMap(reportId, city, {
        goal: goal,
        lines: astroData.lines,
        powerZones: astroData.powerZones,
        birthLocation: birthLocation
      });
    }
    
    // 6. Cities to Avoid Maps
    maps.avoidCityMaps = {};
    const avoidCities = astroData.avoidCities || [];
    
    for (const city of avoidCities.slice(0, 5)) {
      const cityKey = city.name.toLowerCase().replace(/\s+/g, '_');
      maps.avoidCityMaps[cityKey] = await this.generateCityMap(reportId, city, {
        goal: goal,
        lines: astroData.lines,
        powerZones: astroData.powerZones,
        birthLocation: birthLocation,
        isAvoidCity: true
      });
    }
    
    // 7. Legend Page
    maps.legend = await this.generateLegend(reportId);
    
    return { reportId, maps };
  }
  
  // Generate single map and save to temp file
  async generateMap(reportId, mapName, options) {
    const filename = `${reportId}_${mapName}.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = await this.mapRenderer.renderGoalFilteredMap(options);
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`
    };
  }
  
  // Generate city close-up map
  async generateCityMap(reportId, city, options) {
    const { getCityView } = require('./map-renderer');
    const viewConfig = getCityView(city);
    
    const filename = `${reportId}_city_${city.name.toLowerCase().replace(/\s+/g, '_')}.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = await this.mapRenderer.renderGoalFilteredMap({
      ...options,
      viewConfig: viewConfig,
      cities: [{
        ...city,
        type: options.isAvoidCity ? 'avoid' : 'recommended',
        showLabel: true
      }]
    });
    
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`,
      cityName: city.name
    };
  }
  
  // Generate legend page
  async generateLegend(reportId) {
    const filename = `${reportId}_legend.png`;
    const filepath = path.join(this.tempDir, filename);
    
    const buffer = this.legendRenderer.render();
    fs.writeFileSync(filepath, buffer);
    
    return {
      filename,
      filepath,
      base64: buffer.toString('base64'),
      dataUrl: `data:image/png;base64,${buffer.toString('base64')}`
    };
  }
  
  // Clean up temp files after PDF generation
  cleanupTempFiles(reportId) {
    const files = fs.readdirSync(this.tempDir);
    files.forEach(file => {
      if (file.startsWith(reportId)) {
        fs.unlinkSync(path.join(this.tempDir, file));
      }
    });
  }
}

module.exports = ReportGenerator;

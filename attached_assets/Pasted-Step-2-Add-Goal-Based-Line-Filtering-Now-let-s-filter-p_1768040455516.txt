Add Filter Function to Map Renderer
Add this method to AstroMapRenderer class:
javascript// Filter and style lines based on goal
filterLinesByGoal(lines, goal) {
  const config = GOAL_LINE_CONFIG[goal];
  if (!config) return lines; // No filtering if goal not found
  
  return lines.map(line => {
    const isPrimary = config.primary.includes(line.planet);
    const isSecondary = config.secondary.includes(line.planet);
    const isRelevantLineType = config.lineTypes.includes(line.line_type);
    
    let importance;
    if (isPrimary && isRelevantLineType) {
      importance = 'primary';
    } else if (isSecondary && isRelevantLineType) {
      importance = 'secondary';
    } else if (isPrimary || isSecondary) {
      importance = 'secondary';
    } else {
      importance = 'other';
    }
    
    return {
      ...line,
      importance,
      style: LINE_STYLES[importance]
    };
  });
}

// Render map with goal filtering
async renderGoalFilteredMap(options) {
  const { goal, lines, ...restOptions } = options;
  
  // Filter and style lines
  const filteredLines = this.filterLinesByGoal(lines, goal);
  
  // Sort so primary lines render on top
  const sortedLines = filteredLines.sort((a, b) => {
    const order = { other: 0, secondary: 1, primary: 2 };
    return order[a.importance] - order[b.importance];
  });
  
  return this.renderMap({
    ...restOptions,
    lines: sortedLines,
    title: `${goal} - Planetary Influences`
  });
}
Update drawPlanetaryLines to Use Styles
Modify the drawPlanetaryLines method:
javascriptdrawPlanetaryLines(lines, projection) {
  lines.forEach(line => {
    const color = line.color || PLANET_COLORS[line.planet] || '#FFFFFF';
    const points = line.points || line.coordinates;
    const style = line.style || LINE_STYLES.primary;
    
    if (!points || points.length < 2) return;
    
    this.ctx.strokeStyle = color;
    this.ctx.lineWidth = style.lineWidth;
    this.ctx.globalAlpha = style.opacity;
    this.ctx.lineCap = 'round';
    this.ctx.lineJoin = 'round';
    
    // Add glow for primary lines
    if (style.glow) {
      this.ctx.shadowColor = color;
      this.ctx.shadowBlur = 6;
    } else {
      this.ctx.shadowBlur = 0;
    }
    
    this.ctx.beginPath();
    
    points.forEach((point, i) => {
      const lng = point.longitude ?? point[1];
      const lat = point.latitude ?? point[0];
      const projected = projection([lng, lat]);
      
      if (projected) {
        if (i === 0) {
          this.ctx.moveTo(projected[0], projected[1]);
        } else {
          this.ctx.lineTo(projected[0], projected[1]);
        }
      }
    });
    
    this.ctx.stroke();
    this.ctx.globalAlpha = 1;
    this.ctx.shadowBlur = 0;
  });
}
